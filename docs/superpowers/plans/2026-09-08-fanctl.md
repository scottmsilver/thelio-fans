# fanctl Implementation Plan

> **Superseded 2026-09-08:** the controller was reimplemented in Rust at the user's request
> (`fanctl/` crate). The task breakdown, tests and behaviours below were carried over one for
> one; the Python code listings are historical. The spec is the source of truth.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An unprivileged systemd service that holds the Thelio Mira's intake fan at a smooth, never-stopping duty derived from max(CPU package, GPU) temperature, replacing system76-power's on/off cycling.

**Architecture:** A pure `Controller` (EMA, deadband, curve, spin-down delay, decrease slew, re-assert) is driven once a second by a small loop that reads sensors through the existing `fanwatch` package and writes one sysfs file, `pwm2` on the `system76_io` hwmon device. A udev rule makes that file group-writable; the service runs as user `fanctl` with an empty capability set. There is no auto mode in this driver, so every exit path writes a safe duty first.

**Tech Stack:** Python 3.12, standard library plus the existing `nvidia-ml-py` dependency; uv, pytest, ruff, mypy strict; systemd 255; udev.

**Spec:** `docs/superpowers/specs/2026-09-08-fanctl-design.md`

## Global Constraints

- No git commits: this directory is not a git repository and the user has said not to commit.
- `fanwatch` stays read-only; all writes live in the new `fanctl` package.
- Root is used only at install time, by the user running `packaging/install.sh` with sudo.
- Defaults from the spec: curve (50,25) (58,30) (65,40) (75,55) (82,75) (88,100); floor 25 %; EMA 8 s; deadband 2 °C; spin-down delay 30 s; max decrease 3 %/s; re-assert 60 s; sensor failures to panic 5; write failures to exit 10; exit duty 50 %.
- After each task: `uv run pytest`, `uv run ruff check .`, `uv run ruff format .`, `uv run mypy` all clean.
- The service runs from `/opt/fanctl`, an installed copy of this package, because `ProtectHome=yes` hides `/home`.

```python
# src/fanctl/config.py
"""Controller settings: defaults from the design spec, optional TOML overrides, validation."""

from __future__ import annotations

import tomllib
from dataclasses import dataclass, fields, replace
from pathlib import Path

Point = tuple[float, float]


class ConfigError(ValueError):
    """The configuration file is unusable; refuse to start rather than guess."""


@dataclass(frozen=True, slots=True)
class Config:
    curve: tuple[Point, ...] = ((50, 25), (58, 30), (65, 40), (75, 55), (82, 75), (88, 100))
    floor_pct: float = 25
    ema_seconds: float = 8
    deadband_c: float = 2
    spindown_delay_s: float = 30
    max_decrease_pct_per_s: float = 3
    reassert_s: float = 60
    sensor_failures_to_panic: int = 5
    write_failures_to_exit: int = 10
    exit_duty_pct: float = 50
    interval_s: float = 1
    state_path: str = "/run/fanctl/state.json"


def _check(cfg: Config) -> None:
    if len(cfg.curve) < 2:
        raise ConfigError("curve needs at least two points")
    temps = [t for t, _ in cfg.curve]
    duties = [d for _, d in cfg.curve]
    if any(b <= a for a, b in zip(temps, temps[1:], strict=True)):
        raise ConfigError("curve temperatures must strictly increase")
    if any(b < a for a, b in zip(duties, duties[1:], strict=True)):
        raise ConfigError("curve duties must not decrease")
    if not all(0 <= d <= 100 for d in duties):
        raise ConfigError("curve duties must be between 0 and 100")
    if not 0 <= cfg.floor_pct <= 100 or not 0 <= cfg.exit_duty_pct <= 100:
        raise ConfigError("floor_pct and exit_duty_pct must be between 0 and 100")
    for name in ("ema_seconds", "spindown_delay_s", "reassert_s"):
        if getattr(cfg, name) <= 0:
            raise ConfigError(f"{name} must be positive")
    if cfg.deadband_c < 0 or cfg.max_decrease_pct_per_s <= 0:
        raise ConfigError("deadband_c must be >= 0 and max_decrease_pct_per_s > 0")
    if not 0.2 <= cfg.interval_s <= 60:
        raise ConfigError("interval_s must be between 0.2 and 60 seconds")
    if cfg.sensor_failures_to_panic < 1 or cfg.write_failures_to_exit < 1:
        raise ConfigError("failure counts must be at least 1")


def _coerce(name: str, value: object) -> object:
    if name == "curve":
        if not isinstance(value, list) or not all(
            isinstance(p, list) and len(p) == 2 and all(isinstance(v, int | float) for v in p)
            for p in value
        ):
            raise ConfigError("curve must be a list of [temperature, duty] pairs")
        return tuple((float(t), float(d)) for t, d in value)
    if name == "state_path":
        if not isinstance(value, str):
            raise ConfigError("state_path must be a string")
        return value
    if name in ("sensor_failures_to_panic", "write_failures_to_exit"):
        if not isinstance(value, int) or isinstance(value, bool):
            raise ConfigError(f"{name} must be an integer")
        return value
    if not isinstance(value, int | float) or isinstance(value, bool):
        raise ConfigError(f"{name} must be a number")
    return float(value)


def load(path: Path | None) -> Config:
    """Defaults, overridden by the TOML file at `path` if it exists. Raises ConfigError."""
    cfg = Config()
    if path is None or not path.exists():
        _check(cfg)
        return cfg
    try:
        data = tomllib.loads(path.read_text())
    except (OSError, tomllib.TOMLDecodeError) as exc:
        raise ConfigError(f"{path}: {exc}") from exc
    known = {f.name for f in fields(Config)}
    unknown = set(data) - known
    if unknown:
        raise ConfigError(f"{path}: unknown keys {sorted(unknown)}")
    cfg = replace(cfg, **{k: _coerce(k, v) for k, v in data.items()})
    _check(cfg)
    return cfg
```

```python
# src/fanctl/controller.py
"""The control law. Pure: no clocks, no files. Everything comes in through step().

Behaviours borrowed, with thanks, from other fan controllers (behaviour only, no code):
linear point curve and spin-down delay from LACT (MIT); exponential smoothing from
fan2go and CoolerControl; temperature deadband from LACT and CoolerControl; decrease-only
slew limit and periodic re-assert from CoolerControl; missing sensor means full speed
from CoolerControl.
"""

from __future__ import annotations

import math
from dataclasses import dataclass

from fanctl.config import Config, Point

PANIC_TEMP_C = 100.0


@dataclass(frozen=True, slots=True)
class Decision:
    duty_pct: float
    pwm: int
    write: bool
    reason: str
    smoothed_c: float | None
    panic: bool


def interpolate(curve: tuple[Point, ...], temp_c: float) -> float:
    """Linear interpolation over (temperature, duty) points, clamped at both ends."""
    if temp_c <= curve[0][0]:
        return curve[0][1]
    for (t0, d0), (t1, d1) in zip(curve, curve[1:], strict=True):
        if temp_c <= t1:
            return d0 + (d1 - d0) * (temp_c - t0) / (t1 - t0)
    return curve[-1][1]


def pct_to_pwm(pct: float) -> int:
    return max(0, min(255, round(pct * 255 / 100)))


class Controller:
    def __init__(self, cfg: Config) -> None:
        self.cfg = cfg
        self.ema: float | None = None
        self.applied_temp: float | None = None
        self.target_pct: float = cfg.floor_pct
        self.applied_pct: float = cfg.floor_pct
        self.last_time: float | None = None
        self.last_increase_at: float = -math.inf
        self.last_write_at: float = -math.inf
        self.last_written_pwm: int | None = None
        self.sensor_failures = 0

    def _smooth(self, temp_c: float | None, dt: float) -> tuple[float | None, bool]:
        if temp_c is None:
            self.sensor_failures += 1
            if self.sensor_failures >= self.cfg.sensor_failures_to_panic:
                return PANIC_TEMP_C, True
            return self.ema, False
        self.sensor_failures = 0
        if self.ema is None or dt <= 0:
            self.ema = temp_c
        else:
            alpha = 1 - math.exp(-dt / self.cfg.ema_seconds)
            self.ema += alpha * (temp_c - self.ema)
        return self.ema, False

    def step(self, temp_c: float | None, now: float, readback_pwm: int | None) -> Decision:
        dt = 0.0 if self.last_time is None else max(0.0, now - self.last_time)
        self.last_time = now
        smoothed, panic = self._smooth(temp_c, dt)
        reasons: list[str] = []

        # Deadband: re-evaluate the curve only when the smoothed temperature has moved enough.
        if smoothed is not None and (
            panic
            or self.applied_temp is None
            or abs(smoothed - self.applied_temp) >= self.cfg.deadband_c
        ):
            self.target_pct = max(self.cfg.floor_pct, interpolate(self.cfg.curve, smoothed))
            self.applied_temp = smoothed

        # Move the applied duty toward the target: up at once, down slowly and not too soon.
        if self.target_pct > self.applied_pct:
            self.applied_pct = self.target_pct
            self.last_increase_at = now
            reasons.append("increase")
        elif self.target_pct < self.applied_pct:
            if now - self.last_increase_at < self.cfg.spindown_delay_s:
                reasons.append("spin-down delay")
            else:
                budget = self.cfg.max_decrease_pct_per_s * dt
                self.applied_pct = max(self.target_pct, self.applied_pct - budget)
                reasons.append("decrease")

        pwm = pct_to_pwm(self.applied_pct)
        write = False
        if pwm != self.last_written_pwm:
            write = True
            reasons.append("changed")
        elif readback_pwm is not None and readback_pwm != self.last_written_pwm:
            write = True
            reasons.append("readback mismatch")
        elif now - self.last_write_at >= self.cfg.reassert_s:
            write = True
            reasons.append("reassert")
        if write:
            self.last_written_pwm = pwm
            self.last_write_at = now
        if panic:
            reasons.insert(0, "sensor missing: full speed")
        return Decision(
            duty_pct=self.applied_pct,
            pwm=pwm,
            write=write,
            reason=", ".join(reasons) or "steady",
            smoothed_c=smoothed,
            panic=panic,
        )
```

```python
# src/fanctl/pwm.py
"""The one file this program writes: the intake channel's PWM attribute on the Io board."""

from __future__ import annotations

from pathlib import Path

DEVICE_NAME = "system76_io"


def find_device(root: Path, name: str = DEVICE_NAME) -> Path | None:
    """The hwmon directory whose `name` attribute matches, or None. Scanned fresh each call
    because the Io board renumbers when it re-enumerates on USB."""
    for hw in sorted((root / "class/hwmon").glob("hwmon*")):
        try:
            if (hw / "name").read_text().strip() == name:
                return hw
        except OSError:
            continue
    return None


class Pwm:
    def __init__(self, root: Path = Path("/sys"), channel: str = "pwm2") -> None:
        self.root = root
        self.channel = channel
        self._path: Path | None = None

    @property
    def path(self) -> Path | None:
        if self._path is None or not self._path.exists():
            device = find_device(self.root)
            self._path = device / self.channel if device else None
        return self._path

    def write(self, value: int) -> None:
        if not 0 <= value <= 255:
            raise ValueError(f"pwm value {value} out of range 0-255")
        path = self.path
        if path is None:
            raise FileNotFoundError(f"{DEVICE_NAME} {self.channel} not present")
        try:
            path.write_text(str(value))
        except OSError:
            self._path = None  # force a re-scan next time
            raise

    def read(self) -> int | None:
        path = self.path
        if path is None:
            return None
        try:
            return int(path.read_text().strip())
        except (OSError, ValueError):
            return None

    def describe(self) -> str:
        return str(self.path) if self.path else f"{DEVICE_NAME} {self.channel}: not present"
```

```python
# src/fanctl/sensors.py
"""Temperature inputs: CPU package from coretemp in sysfs, GPU through NVML."""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from fanwatch.gpu import Gpu, read_gpu


@dataclass(frozen=True, slots=True)
class Temps:
    cpu_c: float | None
    gpu_c: float | None

    @property
    def max_c(self) -> float | None:
        present = [t for t in (self.cpu_c, self.gpu_c) if t is not None]
        return max(present) if present else None


def _millidegrees(path: Path) -> float | None:
    try:
        return int(path.read_text().strip()) / 1000
    except (OSError, ValueError):
        return None


def cpu_package_c(root: Path) -> float | None:
    """The coretemp package reading; if it has no package sensor, the hottest core."""
    for hw in sorted((root / "class/hwmon").glob("hwmon*")):
        try:
            if (hw / "name").read_text().strip() != "coretemp":
                continue
        except OSError:
            continue
        cores: list[float] = []
        for label in sorted(hw.glob("temp*_label")):
            text = label.read_text().strip() if label.exists() else ""
            value = _millidegrees(label.with_name(label.name.replace("_label", "_input")))
            if value is None:
                continue
            if text.startswith("Package"):
                return value
            cores.append(value)
        if cores:
            return max(cores)
    return None


def read_temps(root: Path = Path("/sys"), gpu_reader: Callable[[], Gpu | None] = read_gpu) -> Temps:
    gpu = gpu_reader()
    return Temps(cpu_c=cpu_package_c(root), gpu_c=None if gpu is None else gpu.temp_c)
```

```python
# src/fanctl/notify.py
"""Minimal sd_notify: one datagram to $NOTIFY_SOCKET. No dependency on libsystemd."""

from __future__ import annotations

import os
import socket


def sd_notify(message: str, socket_path: str | None = None) -> bool:
    """Send one notification. Returns False when there is no socket or the send fails."""
    target = socket_path if socket_path is not None else os.environ.get("NOTIFY_SOCKET")
    if not target:
        return False
    if target.startswith("@"):
        target = "\0" + target[1:]  # abstract namespace
    try:
        with socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM) as sock:
            sock.connect(target)
            sock.send(message.encode())
        return True
    except OSError:
        return False
```

```python
"""A small JSON status file for the dashboard, written atomically each tick."""

from __future__ import annotations

import json
import os
from pathlib import Path


def write_state(path: Path, payload: dict[str, object]) -> None:
    """Write via a temp file and rename. Failure to write status must never stop control."""
    tmp = path.with_suffix(path.suffix + ".tmp")
    try:
        tmp.write_text(json.dumps(payload, sort_keys=True))
        os.replace(tmp, path)
    except OSError:
        try:
            tmp.unlink(missing_ok=True)
        except OSError:
            pass


def read_state(path: Path) -> dict[str, object] | None:
    try:
        data = json.loads(path.read_text())
    except (OSError, ValueError):
        return None
    return data if isinstance(data, dict) else None
```

```python
# src/fanctl/cli.py
"""fanctl: hold the Thelio intake fan at a smooth duty. Runs unprivileged as a systemd service.

    fanctl run        the service loop (Type=notify, watchdog pings, safe duty on exit)
    fanctl dry-run    same loop, prints decisions, writes nothing
    fanctl check      print resolved config, device path, permissions, sensor readings
"""

from __future__ import annotations

import argparse
import os
import signal
import sys
import time
from collections.abc import Callable, Sequence
from dataclasses import asdict
from pathlib import Path
from types import FrameType

from fanctl.config import Config, ConfigError, load
from fanctl.controller import Controller, Decision, pct_to_pwm
from fanctl.notify import sd_notify
from fanctl.pwm import Pwm
from fanctl.sensors import Temps, read_temps
from fanctl.state import write_state
from fanwatch.gpu import Gpu, read_gpu

DEFAULT_CONFIG = Path("/etc/fanctl.toml")


class Service:
    def __init__(
        self,
        cfg: Config,
        *,
        root: Path = Path("/sys"),
        gpu_reader: Callable[[], Gpu | None] = read_gpu,
        dry_run: bool,
    ) -> None:
        self.cfg = cfg
        self.root = root
        self.gpu_reader = gpu_reader
        self.dry_run = dry_run
        self.controller = Controller(cfg)
        self.pwm = Pwm(root)
        self.state_path = Path(cfg.state_path)
        self.write_failures = 0
        self.last_temps = Temps(None, None)
        self.last_error: str | None = None

    def tick(self, now: float) -> Decision:
        temps = read_temps(self.root, self.gpu_reader)
        self.last_temps = temps
        readback = self.pwm.read()
        decision = self.controller.step(temps.max_c, now, readback)
        if decision.write and not self.dry_run:
            try:
                self.pwm.write(decision.pwm)
                self.write_failures = 0
                self.last_error = None
            except OSError as exc:
                self.write_failures += 1
                self.last_error = str(exc)
                if self.write_failures >= self.cfg.write_failures_to_exit:
                    raise RuntimeError(
                        f"pwm write failed {self.write_failures} times: {exc}"
                    ) from exc
        write_state(
            self.state_path,
            {
                "time": time.time(),
                "cpu_c": temps.cpu_c,
                "gpu_c": temps.gpu_c,
                "smoothed_c": decision.smoothed_c,
                "duty_pct": round(decision.duty_pct, 1),
                "pwm": decision.pwm,
                "reason": decision.reason,
                "panic": decision.panic,
                "dry_run": self.dry_run,
                "device": self.pwm.describe(),
                "error": self.last_error,
            },
        )
        return decision

    def shutdown(self) -> None:
        """No auto mode exists in this driver, so leave the fan at a safe duty."""
        if self.dry_run:
            return
        try:
            self.pwm.write(pct_to_pwm(self.cfg.exit_duty_pct))
        except OSError as exc:
            print(f"fanctl: could not set exit duty: {exc}", file=sys.stderr)


def _loop(svc: Service, ticks: int | None, verbose: bool) -> int:
    stop = False

    def on_signal(signum: int, frame: FrameType | None) -> None:
        nonlocal stop
        stop = True

    for sig in (signal.SIGTERM, signal.SIGINT):
        signal.signal(sig, on_signal)
    sd_notify("READY=1")
    done = 0
    try:
        while not stop and (ticks is None or done < ticks):
            started = time.monotonic()
            d = svc.tick(started)
            done += 1
            if verbose:
                t = svc.last_temps
                cpu = "?" if t.cpu_c is None else f"{t.cpu_c:.0f}"
                gpu = "?" if t.gpu_c is None else f"{t.gpu_c:.0f}"
                ema = "?" if d.smoothed_c is None else f"{d.smoothed_c:.1f}"
                mark = "WRITE" if d.write else "     "
                print(
                    f"{time.strftime('%H:%M:%S')} cpu {cpu} gpu {gpu} ema {ema}"
                    f"  -> {d.duty_pct:.0f}% pwm {d.pwm} {mark} {d.reason}",
                    flush=True,
                )
            sd_notify("WATCHDOG=1")
            if ticks is None or done < ticks:
                time.sleep(max(0.0, svc.cfg.interval_s - (time.monotonic() - started)))
    except RuntimeError as exc:
        print(f"fanctl: {exc}", file=sys.stderr)
        svc.shutdown()
        return 1
    finally:
        sd_notify("STOPPING=1")
    svc.shutdown()
    return 0


def _check(cfg: Config, root: Path) -> int:
    print("config:")
    for key, value in asdict(cfg).items():
        print(f"  {key} = {value}")
    pwm = Pwm(root)
    path = pwm.path
    print(f"device: {pwm.describe()}")
    if path is not None:
        st = path.stat()
        mode = oct(st.st_mode & 0o777)
        print(f"  mode {mode} uid {st.st_uid} gid {st.st_gid} writable-by-me {os.access(path, os.W_OK)}")
        print(f"  current pwm {pwm.read()}")
    temps = read_temps(root, read_gpu)
    print(f"sensors: cpu {temps.cpu_c} gpu {temps.gpu_c} max {temps.max_c}")
    print(f"running as uid {os.getuid()} gid {os.getgid()} groups {os.getgroups()}")
    return 0


def parse_args(argv: Sequence[str] | None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(prog="fanctl", description=__doc__,
                                     formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("--config", type=Path, default=DEFAULT_CONFIG)
    parser.add_argument("--sysfs", type=Path, default=Path("/sys"), help=argparse.SUPPRESS)
    parser.add_argument("--state", type=Path, default=None, help="override state file path")
    sub = parser.add_subparsers(dest="command", required=True)
    for name in ("run", "dry-run"):
        p = sub.add_parser(name)
        p.add_argument("--ticks", type=int, default=None, help="stop after N ticks")
    sub.add_parser("check")
    return parser.parse_args(argv)


def main(argv: Sequence[str] | None = None) -> int:
    args = parse_args(argv)
    try:
        cfg = load(args.config)
    except ConfigError as exc:
        print(f"fanctl: config error: {exc}", file=sys.stderr)
        return 2
    if args.state is not None:
        cfg = Config(**{**asdict(cfg), "state_path": str(args.state)})
    if args.command == "check":
        return _check(cfg, args.sysfs)
    dry = args.command == "dry-run"
    svc = Service(cfg, root=args.sysfs, dry_run=dry)
    return _loop(svc, args.ticks, verbose=dry or bool(os.environ.get("FANCTL_VERBOSE")))


if __name__ == "__main__":
    raise SystemExit(main())
```
